-- APFC: automatic sequential numbers for stock transactions -- Safe to run again. This adds a column and trigger; it does not delete stock data. alter table public.stock_transactions add column if not exists transaction_number text; create sequence if not exists public.stock_transaction_number_seq start with 1 increment by 1 no minvalue no maxvalue cache 1; -- Continue after the highest existing APFC transaction number, if any. select setval( 'public.stock_transaction_number_seq', coalesce( (select max(substring(transaction_number from '^ST-([0-9]+)$')::bigint) from public.stock_transactions where transaction_number ~ '^ST-[0-9]+$'), 0 ) + 1, false ); create or replace function public.apfc_set_stock_transaction_number() returns trigger language plpgsql as $$ begin if new.transaction_number is null or btrim(new.transaction_number) = '' then new.transaction_number := 'ST-' || lpad(nextval('public.stock_transaction_number_seq')::text, 6, '0'); end if; return new; end; $$; drop trigger if exists apfc_stock_transaction_number_trigger on public.stock_transactions; create trigger apfc_stock_transaction_number_trigger before insert on public.stock_transactions for each row execute function public.apfc_set_stock_transaction_number(); create unique index if not exists stock_transactions_transaction_number_uidx on public.stock_transactions (transaction_number) where transaction_number is not null;
